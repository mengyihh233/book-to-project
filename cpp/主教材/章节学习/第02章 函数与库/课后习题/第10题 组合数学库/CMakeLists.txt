cmake_minimum_required(VERSION 3.16)
project(roberts_ch02_ex10 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(lesson_library STATIC combinatorics.cpp)
target_include_directories(lesson_library PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(reference reference.cpp)
target_link_libraries(reference PRIVATE lesson_library)
add_executable(contracts tests.cpp)
target_link_libraries(contracts PRIVATE lesson_library)
enable_testing()
add_test(NAME library_contracts COMMAND contracts)
