cmake_minimum_required(VERSION 3.16)
project(RobertsRational LANGUAGES CXX)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use a separate build directory: cmake -S <source> -B <build>")
endif()
option(BOOK_USE_REFERENCE "Build supplied reference instead of your work" OFF)
if(BOOK_USE_REFERENCE)
    set(IMPLEMENTATION 参考实现/rational.cpp)
    message(STATUS "REFERENCE mode: does not count as your exercise completion")
else()
    set(IMPLEMENTATION 我的作答/rational.cpp)
    message(STATUS "WORK mode: unfinished TODOs are expected to fail at runtime")
endif()
add_library(rational STATIC ${IMPLEMENTATION})
target_include_directories(rational PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(rational PUBLIC cxx_std_17)
set_target_properties(rational PROPERTIES CXX_EXTENSIONS OFF)
add_executable(rational_demo demo.cpp)
target_link_libraries(rational_demo PRIVATE rational)
include(CTest)
if(BUILD_TESTING)
    add_executable(rational_checks tests.cpp)
    target_link_libraries(rational_checks PRIVATE rational)
    add_test(NAME rational_contract COMMAND rational_checks)
    set_tests_properties(rational_contract PROPERTIES TIMEOUT 10)
endif()
foreach(target rational rational_demo)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
if(TARGET rational_checks AND NOT MSVC)
    target_compile_options(rational_checks PRIVATE -Wall -Wextra -Wpedantic)
endif()
